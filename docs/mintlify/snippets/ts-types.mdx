# Types TypeScript partagés

Ces interfaces couvrent les principaux types de l'API KORA BUS V1. Importez-les dans vos pages de référence ou copiez-les dans votre projet.

## Types de base

```typescript
// Erreur API standard
export interface KoraApiError {
  error: {
    code: string;
    message: string;
    [key: string]: unknown;
  };
}

// Réponse de succès (enveloppe data)
export interface KoraResponse<T> {
  data: T;
}
```

  verification_token: string;

  platform: 'ios' | 'android';

  email: string;

export interface OAuthTokenRequest {
  provider: 'google' | 'facebook' | 'tiktok';

export interface UpdateProfileRequest {
  name?: string;

  birth_date?: string;

export interface KoraUser {
  uuid: string;
  name: string;
  email: string;

  country_code: string | null;
  birth_date: string | null;

  status: 'active' | 'suspended';

export interface KoraDevice {
  uuid: string;

export interface KoraPreferences {

export interface KoraQuoteRequest {
  departure_uuid: string;

export interface KoraQuote {
  uuid: string;
  departure_uuid: string;

  coefficient: number;
  final_amount: number;

  total_amount: number;

    fare_uuid: string;
    line_code: string;
    origin_stop: string;
    destination_stop: string;

  };

export interface KoraHoldRequest {
  departure_uuid: string;
  origin_stop_uuid: string;
  destination_stop_uuid: string;
  price_quote_uuid: string;
  seats: { seat_uuid: string }[];
}

export interface KoraSegment {
  uuid: string;
  seat_uuid: string;
  seat_code: string;
  origin_stop: string;
  destination_stop: string;

}

export type ReservationStatus = 'held' | 'confirmed' | 'paid' | 'ticketed' | 'boarded' | 'expired' | 'cancelled';

export interface KoraPassenger {

  last_name: string;
  phone: string;
  document_type?: string;
  document_number?: string;
  nationality?: string;
  birth_date?: string;

}

export interface StorePassengersRequest {

    first_name: string;
    last_name: string;
    phone: string;
    seat_uuid: string;
    document_type?: string;
    document_number?: string;
    nationality?: string;
    birth_date?: string;
  }[];
}

export interface KoraReservation {
  uuid: string;
  status: ReservationStatus;
  expires_at: string | null;
  total_amount: number;
  currency: string;
  channel: 'mobile' | 'web' | 'agency';
  departure_uuid: string;
  price_quote_uuid: string;
  passenger_count: number;
  segments: KoraSegment[];

}

export interface KoraPaymentInitiateRequest {

export interface KoraPayment {
  uuid: string;
  status: PaymentStatus;
  amount: number;
  currency: string;
  provider: string;
  provider_reference: string | null;
  confirmed_at: string | null;
  expires_at: string | null;
  polling_interval_seconds: number;

    uuid: string;
    status: ReservationStatus;

  };

export interface PaymentConfirmRequest {
  status: 'success' | 'confirmed' | 'failed';

  provider_payload?: Record<string, unknown>;
}
